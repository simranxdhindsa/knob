
// Added by ios/Widget/build-standby.sh: the effect clock widgets turn their hands with, which
// WidgetKit has but doesn't list. iOS runs it by itself, so the record needs no updates to spin.
public import Foundation

public struct _ClockHandRotationEffect {
  public enum Period {
    case hourHand
    case minuteHand
    case secondHand
    case custom(Swift.Double)
  }
}

extension SwiftUI.View {
  public func _clockHandRotationEffect(_ period: WidgetKit._ClockHandRotationEffect.Period, in timeZone: Foundation.TimeZone, anchor: SwiftUI.UnitPoint) -> some SwiftUI.View
}
